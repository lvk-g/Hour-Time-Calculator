    <script>
        document.addEventListener('DOMContentLoaded', () => {
            // Get DOM elements
            const startTimeInput = document.getElementById('startTime');
            const endTimeInput = document.getElementById('endTime');
            const breakTimeInput = document.getElementById('breakTime');
            const resultDecimal = document.getElementById('resultDecimal');
            const resultExact = document.getElementById('resultExact');
            const errorMessage = document.getElementById('errorMessage');
            const resetBtn = document.getElementById('resetBtn');

            // Set current time as default for start time to be user-friendly
            const now = new Date();
            const currentHour = String(now.getHours()).padStart(2, '0');
            const currentMinute = String(now.getMinutes()).padStart(2, '0');
            // startTimeInput.value = `${currentHour}:${currentMinute}`; // Optional: Un-comment to auto-fill current time

            // Add event listeners to trigger calculation immediately on input change
            [startTimeInput, endTimeInput, breakTimeInput].forEach(input => {
                input.addEventListener('input', calculateTime);
            });

            // Reset functionality
            resetBtn.addEventListener('click', () => {
                startTimeInput.value = '';
                endTimeInput.value = '';
                breakTimeInput.value = '';
                calculateTime();
            });

            // Calculate duration function
            function calculateTime() {
                const startVal = startTimeInput.value;
                const endVal = endTimeInput.value;
                const breakVal = parseInt(breakTimeInput.value) || 0; // default to 0 if empty

                // Hide error message initially
                errorMessage.classList.add('hidden');

                // If either start or end is missing, reset results and return
                if (!startVal || !endVal) {
                    updateDisplay(0, 0, 0);
                    return;
                }

                // Parse hours and minutes from the time strings ("HH:MM")
                const [startH, startM] = startVal.split(':').map(Number);
                const [endH, endM] = endVal.split(':').map(Number);

                // Convert times to total minutes since midnight
                let startTotalMins = (startH * 60) + startM;
                let endTotalMins = (endH * 60) + endM;

                // Handle case where end time is on the next day (e.g. Start 22:00, End 06:00)
                if (endTotalMins < startTotalMins) {
                    endTotalMins += 24 * 60; // Add 24 hours worth of minutes
                }

                // Calculate difference and subtract break
                let durationMins = endTotalMins - startTotalMins - breakVal;

                // Handle invalid case where break is longer than worked time
                if (durationMins < 0) {
                    durationMins = 0;
                    errorMessage.classList.remove('hidden');
                }

                // Convert back to hours and minutes
                const hours = Math.floor(durationMins / 60);
                const minutes = durationMins % 60;
                const decimal = (durationMins / 60).toFixed(2);

                updateDisplay(decimal, hours, minutes);
            }

            // Function to update the DOM with results
            function updateDisplay(decimal, hours, minutes) {
                // Animate numbers (simple scale effect)
                resultDecimal.parentElement.style.transform = 'scale(1.05)';
                setTimeout(() => {
                    resultDecimal.parentElement.style.transform = 'scale(1)';
                }, 150);

                // Update text
                resultDecimal.textContent = parseFloat(decimal).toFixed(2);
                resultExact.textContent = `${hours} Std. ${minutes} Min.`;
            }
        });
    </script>
